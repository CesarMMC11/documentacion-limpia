## Code Example

```javascript title="Node.js (Axios)"
const axios = require('axios');
let data = JSON.stringify({
  "orderId": "3ddc4cfb-c09a-43de-92c1-e4a069732e90",
  "amountReference": "100.001",
  "currencyReference": "USD",
  "allowAmountChange": false,
  "clientIdentification": "V14143800",
  "terminalSerial": "98202003219630",
  "deeplinkConfig": {
    "returnPackage": "com.tuapp.kiosco",
    "returnActivity": "com.tuapp.kiosco.PaymentResultActivity"
  }
});

let config = {
  method: 'get',
  maxBodyLength: Infinity,
  url: '[https://dev.api.spuntodeventa.com/api/v1/pos-terminals/:serialNumber/transactions](https://dev.api.spuntodeventa.com/api/v1/pos-terminals/:serialNumber/transactions)',
  headers: { 
    'Content-Type': 'application/json',
    'x-pos-signature': '<FIRMA_HMAC>'
  },
  data: data
};

axios.request(config)
.then((response) => {
  console.log(JSON.stringify(response.data));
})
.catch((error) => {
  console.log(error);
});
