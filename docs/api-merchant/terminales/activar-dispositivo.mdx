## Code Example

```javascript title="Node.js (Axios)"
const axios = require('axios');
let data = JSON.stringify({
  "serial": "string",
  "code": "string"
});

let config = {
  method: 'post',
  maxBodyLength: Infinity,
  url: '[https://dev.api.spuntodeventa.com/api/v1/pos-terminals/:serialNumber/pairing/activate](https://dev.api.spuntodeventa.com/api/v1/pos-terminals/:serialNumber/pairing/activate)',
  headers: { 
    'Content-Type': 'application/json'
  },
  data: data
};

axios.request(config)
.then((response) => {
  console.log(JSON.stringify(response.data));
})
.catch((error) => {
  console.log(error);
});
