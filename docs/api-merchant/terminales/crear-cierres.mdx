</ul>

## Code Example

```javascript title="Node.js (Axios)"
const axios = require('axios');
let data = JSON.stringify({
  "batchNumber": 3,
  "transactionCount": 12,
  "closedAt": "2023-04-04T15:26:51.187Z",
  "currencyReference": "USD",
  "terminal": {
    "id": "TMS ID",
    "affiliateCode": "0010880050",
    "number": "98202003219630",
    "serial": "98202003219630"
  }
});

let config = {
  method: 'post',
  url: '[https://dev.api.spuntodeventa.com/api/v1/pos-terminals/:serialNumber/settlements](https://dev.api.spuntodeventa.com/api/v1/pos-terminals/:serialNumber/settlements)',
  headers: { 'Content-Type': 'application/json', 'x-pos-signature': '<FIRMA_HMAC>' },
  data: data
};
// ... resto del request axios
