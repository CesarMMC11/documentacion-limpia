## Code Example

```javascript title="Node.js (Axios)"
const axios = require('axios');
let data = JSON.stringify({
  "serial": "string"
});

let config = {
  method: 'post',
  maxBodyLength: Infinity,
  url: '[https://dev.api.spuntodeventa.com/api/v1/merchants/:merchantId/pos-terminals/:serialNumber/pairing](https://dev.api.spuntodeventa.com/api/v1/merchants/:merchantId/pos-terminals/:serialNumber/pairing)',
  headers: { 
    'Content-Type': 'application/json',
    'Authorization': 'Bearer <token>'
  },
  data: data
};

axios.request(config)
.then((response) => {
  console.log(JSON.stringify(response.data));
})
.catch((error) => {
  console.log(error);
});
